import numpy as np

data = [12, 14, 15, 16, 17, 18, 19, 22, 95]
q1, q3 = np.percentile(data, [25, 75])
iqr = q3 - q1
lower, upper = q1 - 1.5 * iqr, q3 + 1.5 * iqr
outliers = [x for x in data if x < lower or x > upper]
print("Outliers:", outliers)

### Program 18
from itertools import combinations

txns = [
    ["Milk", "Bread", "Eggs"],
    ["Milk", "Bread"],
    ["Milk", "Eggs"],
    ["Bread", "Butter"],
    ["Milk", "Bread", "Butter"]
]
min_support = 2
items = sorted(list(set(item for t in txns for item in t)))

frequent = []
for k in range(1, len(items) + 1):
    itemsets = list(combinations(items, k))
    current = [itemset for itemset in itemsets if sum(1 for t in txns if set(itemset).issubset(t)) >= min_support]
    if not current:
        break
    frequent.extend(current)

print("Frequent Itemsets:", frequent)


### Program 19
import multiprocessing as mp

def process_part(arr):
    return sum(x * x for x in arr)

if __name__ == '__main__':
    data = list(range(1, 101))
    mid = len(data) // 2
    part1, part2 = data[:mid], data[mid:]
    
    with mp.Pool(2) as pool:
        res1, res2 = pool.map(process_part, [part1, part2])
        
    print("Part 1 Result:", res1)
    print("Part 2 Result:", res2)


### Program 20
import time
import multiprocessing as mp

def sum_part(arr):
    return sum(arr)

if __name__ == '__main__':
    numbers = list(range(10_000_000))
    
    t0 = time.time()
    seq_sum = sum(numbers)
    seq_time = time.time() - t0
    
    mid = len(numbers) // 2
    t0 = time.time()
    with mp.Pool(2) as pool:
        par_sum = sum(pool.map(sum_part, [numbers[:mid], numbers[mid:]]))
    par_time = time.time() - t0
    
    print(f"Sequential Time: {seq_time:.4f}s")
    print(f"Parallel Time:   {par_time:.4f}s")
